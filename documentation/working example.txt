Here's a complete, minimal, and working example of sending OpenTelemetry traces to an OTLP endpoint with authentication in Python (2025–2026 style):python

import time
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.otlp.proto.http.trace_exporter import OTLPSpanExporter
from opentelemetry.sdk.resources import Resource
from opentelemetry.semconv.resource import ResourceAttributes

# ── 1. Choose your vendor configuration ─────────────────────────────────────

# Option A - Most common style (Grafana Cloud, Honeycomb, many others)
ENDPOINT = "https://otlp-gateway-prod-us-central-0.grafana.net/otlp/v1/traces"
HEADERS = {
    "Authorization": "Basic YOUR_API_KEY_BASE64_HERE"   # usually base64(api_key + ":")
    # or for Honeycomb:    "x-honeycomb-team": "hc-your-long-api-key",
    # or for Elastic:      "Authorization": "Bearer your-elastic-apm-token",
}

# Option B - Very explicit full URL (sometimes needed)
# ENDPOINT = "https://api.honeycomb.io/1/traces"
# HEADERS = {"x-honeycomb-team": "your-honeycomb-api-key"}

# ── 2. Set up resource (helps identify your service) ────────────────────────

resource = Resource(attributes={
    ResourceAttributes.SERVICE_NAME: "my-python-demo-app",
    ResourceAttributes.SERVICE_VERSION: "1.0.0",
    ResourceAttributes.DEPLOYMENT_ENVIRONMENT: "development",
})

# ── 3. Initialize tracing ───────────────────────────────────────────────────

trace.set_tracer_provider(TracerProvider(resource=resource))

# Create OTLP exporter with auth
otlp_exporter = OTLPSpanExporter(
    endpoint=ENDPOINT,
    headers=HEADERS,
    # Optional but very recommended in production:
    compression="gzip",           # reduces bandwidth a lot
    timeout=10,                   # seconds
)

# Batch processor (recommended - sends data in batches)
span_processor = BatchSpanProcessor(otlp_exporter)
trace.get_tracer_provider().add_span_processor(span_processor)

# Get a tracer
tracer = trace.get_tracer(__name__)

# ── 4. Create some example spans ────────────────────────────────────────────

def do_work():
    with tracer.start_as_current_span("important-operation") as span:
        span.set_attribute("http.method", "GET")
        span.set_attribute("user.id", 12345)
        
        with tracer.start_as_current_span("database-query") as db_span:
            db_span.set_attribute("db.system", "postgresql")
            db_span.set_attribute("db.name", "users")
            time.sleep(0.3)  # simulate work
            
        with tracer.start_as_current_span("external-api-call"):
            time.sleep(0.6)  # simulate slow external call

# Run the example a few times
print("Sending test traces...")
for i in range(3):
    with tracer.start_as_current_span(f"request-{i+1}", attributes={"request.id": f"req_{i+1}"}):
        do_work()
    time.sleep(0.8)

# Give the batch processor time to send data
print("Waiting for traces to be sent...")
time.sleep(3)

print("Done! Check your observability backend (Grafana, Honeycomb, etc.)")

Quick checklist before running:Replace ENDPOINT with your real OTLP HTTP endpoint
Replace the HEADERS value with your real API key/token
Most vendors use HTTP/protobuf on port 4318 these days (not gRPC/4317)
Make sure you installed the packages:
